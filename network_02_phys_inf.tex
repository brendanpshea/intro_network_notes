\section{Physical Layer Fundamentals}
\subsection{Signal Transmission}

\articleonly{
Understanding how data travels at the physical layer is fundamental to network troubleshooting and design. In this section, we explore how electrical signals represent data and the key concepts of duplex communication.
}

\begin{frame}{How Does Data Travel on a Wire?}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            \begin{itemize}
                \item Computers represent data as \keyterm{bits}---1s and 0s.
                \item On copper cables, bits travel as \keyterm{electrical voltage changes}.
                \item Higher voltage = 1, lower voltage = 0 (simplified).
                \item \keyterm{Baseband signaling}: The entire cable carries one signal at a time---like a one-lane road.
                \item This is why cable quality matters: poor cables distort the voltage changes, corrupting your data!
            \end{itemize}
            
            \vspace{0.3em}
            \begin{exampleblock}{Analogy}
                Think of Morse code with a flashlight: on = 1, off = 0. Now imagine doing that billions of times per second!
            \end{exampleblock}
        \end{column}
        \begin{column}{0.38\textwidth}
            \begin{center}
                % CAPTION: Digital signal representation showing voltage over time

\subsection{Duplex Communication}

\begin{frame}{Half-Duplex vs Full-Duplex}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Half-Duplex}
                \begin{itemize}
                    \item Only \textbf{one device} can transmit at a time.
                    \item Like a \textbf{walkie-talkie}: ``Over!''
                    \item Must wait for the other person to finish.
                    \item Old \textbf{hubs} forced half-duplex.
                    \item If two devices transmit together = \keyterm{collision}.
                \end{itemize}
                
                \vspace{0.5em}
                \centering
                % CAPTION: Logical OR operation showing two PCs transmitting simultaneously
                \begin{tikzpicture}[scale=0.7]
                    \node[pc] (pc1) at (0,0) {\iconpc[0.4cm]};
                    \node[pc] (pc2) at (3,0) {\iconpc[0.4cm]};
                    \draw[-{Stealth}, line width=2pt, rctcblue] (0.4,0.1) -- (2.6,0.1);
                    \node[font=\tiny, above] at (1.5,0.1) {OR};
                    \draw[{Stealth}-, line width=2pt, rctcblue!50] (0.4,-0.1) -- (2.6,-0.1);
                \end{tikzpicture}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Full-Duplex}
                \begin{itemize}
                    \item \textbf{Both devices} can transmit simultaneously.
                    \item Like a \textbf{phone call}: both people can talk.
                    \item No waiting, no collisions.
                    \item Modern \textbf{switches} enable full-duplex.
                    \item Effectively \textbf{doubles} throughput!
                \end{itemize}
                
                \vspace{0.5em}
                \centering
                \begin{tikzpicture}[scale=0.7]
                    \node[pc] (pc1) at (0,0) {\iconpc[0.4cm]};
                    \node[pc] (pc2) at (3,0) {\iconpc[0.4cm]};
                    \draw[-{Stealth}, line width=2pt, networkgreen] (0.4,0.15) -- (2.6,0.15);
                    \node[font=\tiny, above] at (1.5,0.15) {AND};
                    \draw[{Stealth}-, line width=2pt, networkgreen] (0.4,-0.15) -- (2.6,-0.15);
                \end{tikzpicture}
            \end{block}
        \end{column}
    \end{columns}
    
    \vspace{0.3em}
    \begin{alertblock}{Why This Matters}
        A 1 Gbps connection in full-duplex can send AND receive 1 Gbps simultaneously---twice the effective bandwidth of half-duplex!
    \end{alertblock}
\end{frame}

\subsection{Collision Domains}

\begin{frame}{Collisions: A Problem We Solved}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item A \keyterm{collision} occurs when two devices transmit at the same time---the signals get scrambled.
                \item Old solution: \keyterm{CSMA/CD} (Carrier Sense Multiple Access with Collision Detection).
                \item CSMA/CD means: ``Listen before you talk; if someone else is talking, stop and wait.''
                \item \keyterm{Hubs} share one collision domain---all devices compete for the same ``airtime.''
                \item \keyterm{Switches} give each port its own collision domain---no more collisions!
            \end{itemize}
            
            \vspace{0.3em}
            \begin{exampleblock}{Analogy}
                \scriptsize
                Hub = one conversation in a crowded room (everyone talks over each other). Switch = private phone lines for everyone.
            \end{exampleblock}
        \end{column}

% SLIDE 11: Why Do Faster Speeds Need Better Cables?
% =============================================================================
\begin{frame}{Why Do Faster Speeds Need Better Cables?}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            \begin{itemize}
                \item Faster speeds mean more bits packed into the same amount of time.
                \item It's like speaking faster---you need \textbf{clearer audio} to understand rapid speech.
                \item Higher frequencies (more bits/second) are more vulnerable to:
                    \begin{itemize}
                        \item \keyterm{Attenuation}: Signal gets weaker over distance
                        \item \keyterm{Crosstalk}: Signals bleed between wire pairs
                        \item \keyterm{EMI}: External interference corrupts data
                    \end{itemize}
                \item Better cables have: tighter twists, better shielding, stricter manufacturing standards.
            \end{itemize}
            
            \vspace{0.3em}
            \begin{alertblock}{The Bottom Line}
                \small
                You can't run 10 Gbps on old Cat5 cable---the signal degrades too much!
            \end{alertblock}
        \end{column}
        \begin{column}{0.38\textwidth}
            \begin{center}

\section{Copper Cabling}
\subsection{Twisted Pair Technology}

\articleonly{
Twisted pair cabling is the most common medium for LAN connectivity. The twisting of wire pairs provides protection against electromagnetic interference, making reliable high-speed data transmission possible.
}

\begin{frame}{Why Twisted Pair? The Science of Interference}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \small
                \item Electrical signals create \keyterm{electromagnetic fields} around wires.
                \item Nearby power cables, motors, and fluorescent lights cause \keyterm{EMI (Electromagnetic Interference)}.
                \item EMI can corrupt your data---turning 1s into 0s!
                \item \textbf{The solution}: Twist the wire pairs together.
                \item When wires are twisted, interference affects both wires equally---the receiver can \textbf{subtract out} the noise.
                \item More twists per inch = better protection.
            \end{itemize}
            
            \vspace{0.3em}
            \begin{exampleblock}{Fun Fact}
                \scriptsize
                This technique was invented for telephone lines in the 1880s---it's been protecting signals for over 140 years!
            \end{exampleblock}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{center}
                \mode<beamer>{\includegraphics[width=\textwidth]{twisted_pair_annotated.png}}
                \articleonly{\includegraphics[width=\textwidth]{twisted_pair_annotated.png}\\
                \small\emph{Figure: Annotated twisted-pair cable showing conductor pairs and protective layers.}}
            \end{center}
            
            \vspace{0.3em}
            \begin{block}{Why Twisting Works}
                \small
                Each wire in a pair picks up the \textit{same} interference. The receiver compares the two signals and cancels out anything that's identical---the noise disappears!
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\subsection{Unshielded Twisted Pair (UTP)}

\begin{frame}[t,shrink=12]{Unshielded Twisted Pair (UTP)}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}

                    \item Keep runs under \textbf{90 meters} for horizontal cabling (10m for patch cables)
                    \item Cross power cables at \textbf{90° angles}
                    \item \textbf{Label both ends} immediately
                    \item Test every cable after termination

                \begin{itemize}
                    \item Don't bend cables sharply---damages internal wires
                    \item Don't run parallel to power cables---causes EMI
                    \item Don't exceed 25 lbs pull tension
                    \item Don't untwist pairs more than 0.5 inch at terminations
                    \item Don't leave cables unlabeled
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}

        Total cable path = 100 meters max. Permanent horizontal cabling $\leq$ 90m, patch cables $\leq$ 10m combined.

\section{Fiber Optic Cabling}
\subsection{Introduction to Fiber}

\articleonly{
Fiber optic cabling uses light to transmit data at high speeds over long distances with excellent resistance to electromagnetic interference. In this section, we examine fiber construction, connector types, and practical deployment choices.
}

\begin{frame}{Why Fiber Optic?}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item Copper cables transmit \textbf{electrical signals}---fiber transmits \textbf{pulses of light}.
                \item Light is immune to \keyterm{electromagnetic interference}---no crosstalk, no EMI problems!
                \item Fiber supports \textbf{much longer distances}: kilometers instead of 100 meters.
                \item Higher \textbf{bandwidth potential} for future upgrades.
            \end{itemize}
            
            \vspace{0.3em}
            \begin{alertblock}{Trade-offs}
                More expensive, harder to terminate, requires special tools, more fragile than copper.
            \end{alertblock}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{When to Choose Fiber}
                \begin{itemize}
                    \item Distance over 100 meters
                    \item High EMI environments
                    \item Between buildings
                    \item Need for extreme speed
                    \item Security (harder to tap)
                \end{itemize}
            \end{block}
            
            \vspace{0.3em}
            \begin{exampleblock}{Fun Fact}
                Light in fiber travels at about 200,000 km/sec---fast enough to circle Earth 5 times per second!
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

                \item \keyterm{Cladding}: Surrounds the core; reflects light back inward through \keyterm{total internal reflection}.
                \item \keyterm{Buffer coating}: Protective layer around cladding.
                \item \keyterm{Jacket}: Outer protection (may include strength members).
            \end{itemize}

\subsection{Advanced Fiber Techniques}

\begin{frame}{Fiber Splicing and WDM}

                \begin{itemize}
                    \item \keyterm{Fusion splice}: Melts fibers together permanently---lowest signal loss (0.1 dB)

                \vspace{0.3em}
                \begin{itemize}
                    \item \keyterm{CWDM}: Coarse WDM---up to 18 channels, cheaper
                    \item \keyterm{DWDM}: Dense WDM---80+ channels, expensive
                \end{itemize}
            \end{block}
            
            \vspace{0.3em}
            \begin{exampleblock}{Analogy}
                WDM is like sending red, blue, and green flashlight beams through the same fiber---each carries different data!
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\subsection{Case Study}

\begin{frame}{Case Study: Wednesday's Cemetery Office}
    \begin{casestudy}{Wednesday's Cemetery Office}
        Wednesday is setting up a writing office in the cemetery gatehouse, which is 300 meters from the mansion. She needs a fast, reliable network connection for uploading her novels.
        
        \vspace{0.5em}
        Uncle Fester suggests running Cat6a cable underground through the old electrical conduits (which still have some active 240V lines for the crypt lighting).
        
        \vspace{0.5em}
        Lurch suggests fiber optic cable instead.
    \end{casestudy}
    
    \begin{block}{Review Questions}
        \begin{enumerate}
            \item Can Cat6a reach 300 meters? Why or why not?
            \item What two advantages would fiber have in this scenario?
            \item Should Wednesday use single-mode or multimode fiber?
        \end{enumerate}
    \end{block}
\end{frame}

% =============================================================================
% SLIDE 38: Case Study Solution - Wednesday's Cemetery Office
% =============================================================================
\begin{frame}{Case Study Solution: Wednesday's Cemetery Office}
    \begin{casesolution}{Wednesday's Cemetery Office}
        \begin{enumerate}
            \item \textbf{No}---Cat6a maximum distance is 100 meters. At 300 meters, the signal would be completely unusable (3x over the limit!).
            \item Fiber advantages for this scenario:
                \begin{itemize}
                    \item \textbf{Distance}: Easily handles 300 meters (MMF goes to 550m)
                    \item \textbf{EMI immunity}: Won't be affected by the 240V electrical lines in the conduit
                \end{itemize}
            \item \textbf{Multimode (MMF)} is the better choice:
                \begin{itemize}
                    \item 300m is well within MMF range (up to 550m)
                    \item MMF transceivers cost less than SMF
                    \item No need for SMF's multi-kilometer capability
                \end{itemize}
        \end{enumerate}
    \end{casesolution}
    
    \begin{alertblock}{Key Lesson}
        When copper can't reach and EMI is present, fiber is the answer. Choose MMF for building-scale distances, SMF for campus or beyond.
    \end{alertblock}
\end{frame}

\begin{frame}{Advanced Testing Tools}

        \begin{column}{0.48\textwidth}
            \begin{block}{Copper Testing}
                \keyterm{TDR (Time Domain Reflectometer)}:
                \begin{itemize}
                    \item Sends signal, measures reflection
                    \item Tells you WHERE a break is
                    \item ``The break is 47 meters out''
                \end{itemize}
                
                \vspace{0.3em}
                \keyterm{Certification tester}:
                \begin{itemize}
                    \item Verifies cable meets Cat specs
                    \item Tests attenuation, crosstalk, length
                    \item Required for warranty compliance
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Fiber Testing}
                \keyterm{OTDR (Optical TDR)}:
                \begin{itemize}
                    \item TDR for fiber optic cables
                    \item Finds breaks, splices, connectors
                    \item Shows signal loss at each point
                \end{itemize}
                
                \vspace{0.3em}
                \keyterm{Visual fault locator}:
                \begin{itemize}
                    \item Shines visible red laser
                    \item Breaks glow or leak light
                    \item Quick and inexpensive
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    
\end{frame}

\subsection{Case Study}

\begin{frame}{Case Study: Pugsley's Flickering Connection}
    \begin{casestudy}{Pugsley's Flickering Connection}
        Pugsley's computer keeps losing network connectivity intermittently. The link light on his switch port flickers randomly.
        
        \vspace{0.5em}
        Lurch investigates and finds:
        \begin{itemize}
            \item Basic cable tester shows all 8 wires have continuity---PASS
            \item The cable run is only 40 meters (Cat5e)
            \item When Lurch wiggles the cable near the patch panel, the connection drops
        \end{itemize}
    \end{casestudy}
    
    \begin{block}{Review Questions}
        \begin{enumerate}
            \item If the cable passed continuity testing, what type of problem might this be?
            \item What should Lurch inspect closely at the patch panel?
            \item What tool would definitively identify the issue?
        \end{enumerate}
    \end{block}
\end{frame}

% =============================================================================
% SLIDE 47: Case Study Solution - Pugsley's Flickering Connection
% =============================================================================
\begin{frame}{Case Study Solution: Pugsley's Flickering Connection}
    \begin{casesolution}{Pugsley's Flickering Connection}
        \begin{enumerate}
            \item This is likely a \textbf{bad termination}. The wires make contact (passing continuity) but aren't solidly connected---movement breaks the connection.
            \item At the patch panel, Lurch should look for:
                \begin{itemize}
                    \item Wires not fully seated in the 110 block
                    \item Too much untwisted wire (should be $<$0.5 inch)
                    \item Jacket stripped too far, causing strain on wires
                \end{itemize}
            \item A \textbf{certification tester} would show marginal or failing results for crosstalk and return loss---problems a basic tester misses.
        \end{enumerate}
    \end{casesolution}
    
    \vspace{0.3em}
    \begin{alertblock}{Quick Fix}
        Re-punch the cable at the patch panel. If that doesn't work, re-terminate both ends. Termination problems are the most common cable issues!
    \end{alertblock}
\end{frame}

\section*{Module Summary}

\begin{frame}[t,allowframebreaks]{Module 2.0 Summary}
    \small
    \textbf{Key Concepts:}
    \begin{itemize}
        \item Physical Layer transmits data as electrical signals (copper) or light pulses (fiber).
        \item \keyterm{UTP} (Unshielded Twisted Pair) is standard; \keyterm{STP} (Shielded Twisted Pair) for high-EMI environments.
        \item Cable categories: \keyterm{Cat5e} supports Gigabit Ethernet; \keyterm{Cat6/6a} supports 10 Gigabit Ethernet.
        \item \keyterm{Fiber optic}: \keyterm{MMF} (multimode, orange) for short distances within buildings; \keyterm{SMF} (single-mode, yellow) for long distances between buildings.
        \item Wiring standards: \keyterm{T568A} and \keyterm{T568B}. Use same standard on both ends for straight-through cable.
        \item Common connectors: \keyterm{RJ-45} (copper), \keyterm{LC/SC/ST} (fiber).
    \end{itemize}
    \framebreak
    
    \vspace{0.3em}
    \textbf{Installation \& Troubleshooting:}
    \begin{itemize}
        \item Use \keyterm{plenum}-rated cable in air handling spaces (fire code requirement).
        \item Maximum copper run: 100 meters. Fiber can go much farther (300m MMF, 10+ km SMF).
        \item Common problems: bad terminations, \keyterm{attenuation} (signal loss), \keyterm{crosstalk} (interference).
        \item Tools: cable testers, tone generators, TDR (Time Domain Reflectometer), certification testers.
        \item Always start troubleshooting at Layer 1---if the cable isn't working, nothing else matters!
    \end{itemize}
\end{frame}
