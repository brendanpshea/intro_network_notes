% SLIDE 5: TLS Overview
\begin{frame}{TLS: Transport Layer Security}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Definition}
                \small
                \textbf{TLS} is a cryptographic protocol designed to provide:
                \begin{enumerate}
                    \item \textbf{Encryption}: Privacy (No eavesdropping).
                    \item \textbf{Integrity}: Data has not been changed.
                    \item \textbf{Authentication}: Verifying identity.
                \end{enumerate}
            \end{block}
            \begin{exampleblock}{Analogy: Mac's "Ocular Patdown"}
                \small
                Mac assesses a threat (Authentication), ensures they aren't carrying weapons (Integrity), and clears them for entry.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{center}

% SLIDE 10: PTP
\begin{frame}{Precision Time Protocol (PTP)}
    \begin{block}{When Milliseconds Aren't Good Enough}
        \small
        \textbf{NTP} is accurate to milliseconds (ms). \textbf{PTP (IEEE 1588)} is accurate to \textbf{microseconds ($\mu$s)} or even nanoseconds.
    \end{block}
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Use Cases}
                \small
                \begin{itemize}
                    \item \textbf{5G}: Cellular Sync.
                    \item \textbf{Finance}: High-frequency trading.
                    \item \textbf{Automation}: Robotics.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Analogy: The Nightman Cometh}

% SLIDE 12: Case Study Solution
\begin{frame}{Case Study Solution: The Paddy's Time Heist}
    \begin{casesolution}{"The Gang Learns About Time"}
        \small
        \begin{enumerate}
            \item \textbf{Validity}: The system thinks it is 1970. The certificate (from 2024) is "from the future" and thus invalid.
            \item \textbf{Protocol}: \textbf{NTP} (Network Time Protocol).
            \item \textbf{Fix}: Configure the POS to sync with \texttt{pool.ntp.org}.
        \end{enumerate}
    \end{casesolution}
    
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{The Fix}
                \ttfamily\small
                \# chrony.conf\\
                server 0.pool.ntp.org iburst
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Lesson}
                \small
                If time is broken, \textbf{Security is broken}. (Logs, Auth, and Certs all fail).
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Case Study Solution: The Gang's Call Center}
    \begin{casesolution}{"The Gang Learns QoS"}
        \small
        \begin{enumerate}
            \item \textbf{Issue}: \textbf{Jitter} (Latency variation). Real-time voice cannot tolerate delays.
            \item \textbf{Cause}: \textbf{Congestion}. Mac is filling the bandwidth pipe, forcing voice packets to wait in the buffer.
            \item \textbf{Fix}: Voice and video traffic should be on a separate VLAN. The voice VLAN should implement \textbf{QoS (Quality of Service)} to ensure it has higher priority than best-effort traffic (like game downloads).
        \end{enumerate}
    \end{casesolution}
    
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{QoS Configuration}
                \small
                \begin{itemize}
                    \item \textbf{Tagging}: Mark voice packets with \textbf{DSCP EF (46)} (Expedited Forwarding).
                    \item \textbf{Queuing}: Give EF packets strict priority over Best Effort (Game data).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.6]
                    \draw[thick] (0,0) -- (5,0);
                    \draw[thick] (0,2) -- (5,2);
                    \node[font=\tiny] at (2.5, -0.3) {Network Pipe};

                    \node[font=\tiny] at (1, 1) {Game};
                    
                    % Dee's Voice (Fast Lane)
                    \draw[fill=paddysgreen] (3.5, 0.5) rectangle (4.5, 1.5);
