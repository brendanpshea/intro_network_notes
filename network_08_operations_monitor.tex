% SLIDE 7: Case Study - Donkey's Waffles
\begin{frame}{Case Study: Donkey's Unauthorized Wi-Fi}
    \begin{alertblock}{Scenario: "I'm making Waffles... and Wi-Fi!"}
        \footnotesize
        Donkey wants to stream music in the Swamp. He buys a cheap \$20 Router from Best Buy and plugs it into the main Corporate Switch.
        
        \vspace{0.2em}
        \textbf{The Result:}
        \begin{itemize}
            \item The Router broadcasts "Donkey-Guest" (Open/No Password).
            \item The Router starts handing out bad IP addresses (Rogue DHCP).
            \item Lord Farquaad's spies connect to it and access the internal Castle network.
        \end{itemize}
    \end{alertblock}
    \begin{block}{Discussion}
        \footnotesize
        \begin{enumerate}
            \item Which Change Management steps did Donkey skip?
            \item What technical control could have stopped this immediately?
        \end{enumerate}
    \end{block}
\end{frame}

% SLIDE 8: Case Study Solution
\begin{frame}{Case Study Solution}
    \begin{exampleblock}{The Fix}
        \footnotesize
        \begin{itemize}
            \item \textbf{Process Failure}: Donkey skipped \textbf{Request}, \textbf{Impact Analysis}, and \textbf{Approval}. This was an "Unauthorized Change." No Change Advisory Board (CAB) existed to say "Wait, what are you doing?"
            
            \item \textbf{Technical Fix}: Enable \textbf{Port Security} on the network switch. This feature limits how many MAC addresses (devices) can connect to a single port. When Donkey plugged in his router, it immediately tried to broadcast multiple MAC addresses (one for itself, plus virtual ones for DHCP). The port security rule would have violated the limit and \textbf{shutdown the port automatically}.
            
            \item \textbf{Alternative Control}: Implement \textbf{NAC (Network Access Control)}. This would scan Donkey's unauthorized router, see it is not on the approved device list, and either block it or quarantine it in a "guest" VLAN with restricted access.
            
            \item \textbf{Lesson}: Never skip the Change Management process, and always enforce technical controls like Port Security on access ports where regular users (and Donkeys) can plug things in.
        \end{itemize}
    \end{exampleblock}
\end{frame}

% SLIDE 39: Shaping vs. Policing
\begin{frame}{Managing Bandwidth: Shaping vs. Policing}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Traffic Shaping (The Buffer)}
                \footnotesize
                \textbf{"Please Wait."}
                \begin{itemize}
                    \item Delays excess packets in a queue (Buffer).
                    \item Smooths out bursts.
                    \item \textit{Analogy:} A waiting room.
                \end{itemize}
            \end{block}
            \begin{center}
                \begin{tikzpicture}[scale=0.5]
                    \draw[->, thick] (0,0) -- (4,0) -- (4,2) -- (5,2) -- (5,0) -- (8,0);
                    \node[above, font=\tiny] at (2,0) {Burst};
                    \node[above, font=\tiny] at (6.5,0) {Smooth};
                \end{tikzpicture}
            \end{center}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Traffic Policing (The Chop)}
                \footnotesize
                \textbf{"Get Out!"}
                \begin{itemize}
                    \item Drops excess packets immediately.
                    \item Hard limit.
                    \item \textit{Analogy:} A bouncer cutting the line.
                \end{itemize}
            \end{block}
            \begin{center}
                \begin{tikzpicture}[scale=0.5]
                    \draw[->, thick] (0,0) -- (4,0);
                    \draw[thick, red] (4, -0.5) -- (4, 2);
                    \node[right, font=\tiny, text=red] at (4, 1) {DROP!};
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 40: Marking the Packets (CoS and DSCP)
\begin{frame}{Classification: Stamping the Envelope}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{block}{How does the router know?}
                \footnotesize
                We add a "Tag" to the packet header so routers know it is VIP.
            \end{block}
            \begin{exampleblock}{Layer 2: CoS (Class of Service)}
                \footnotesize
                \begin{itemize}
                    \item Lives in the \textbf{802.1Q VLAN Header}.
                    \item Range: 0 (Best Effort) to 7 (Network Control).
                    \item \textbf{Voice usually = 5}.
                \end{itemize}
            \end{exampleblock}
            \begin{exampleblock}{Layer 3: DSCP (DiffServ Code Point)}
                \footnotesize
                \begin{itemize}
                    \item Lives in the \textbf{IP Header}.
                    \item \textbf{EF (Expedited Forwarding)}: The absolute highest priority for Voice.
                \end{itemize}
            \end{exampleblock}
        \end{column}

    \end{alertblock}

            \item What DSCP tag should the VoIP phones be using?
        \end{enumerate}
    \end{block}

% SLIDE 42: Case Study Solution
\begin{frame}{Case Study Solution: The Robot Voice}
    \begin{exampleblock}{The Fix: Apply QoS}
        \footnotesize
        \begin{enumerate}
            \item \textbf{Root Cause}: \textbf{Jitter}. (Variation in arrival time makes voice sound robotic).
            \item \textbf{Strategy}: Configure \textbf{Queuing} on the router.
            \begin{itemize} \scriptsize
                \item Create a "Low Latency Queue" (LLQ) for Voice.
                \item Put the Movies in the "Best Effort" queue.
            \end{itemize}
            \item \textbf{Tagging}: Ensure VoIP phones mark traffic as \textbf{EF (Expedited Forwarding)} or \textbf{DSCP 46}.
        \end{enumerate}
    \end{exampleblock}
